328 template <typename TO_PRIMITIVE>
329 static GrowableBuffer<TO_PRIMITIVE>
330 copy_as(const GrowableBuffer<PRIMITIVE>& other) {
331 int64_t len = (int64_t)other.length();
332 int64_t actual =
333 (len < other.options_.initial()) ? other.options_.initial() : len;
334
335 if (!awkward::is_tt<std::complex, TO_PRIMITIVE>::value &&
336 awkward::is_tt<std::complex, PRIMITIVE>::value) {
337 len *= 2;
338 actual *= 2;
339 }
340
341 auto ptr =
342 std::unique_ptr<TO_PRIMITIVE[]>(new TO_PRIMITIVE[(size_t)actual]);
343 TO_PRIMITIVE* rawptr = ptr.get();
344
345 other.panel_->copy_as(rawptr, 0);
346
347 return GrowableBuffer<TO_PRIMITIVE>(
348 BuilderOptions(actual, other.options().resize()),
349 std::move(ptr),
350 len,
351 actual);
352 }

455 void
456 extend(const PRIMITIVE* ptr, size_t size) {
457 size_t unfilled_items = ptr_->reserved() - ptr_->current_length();
458 if (size > unfilled_items) {
459 for (size_t i = 0; i < unfilled_items; i++) {
460 fill_panel(ptr[i]);
461 }
462 add_panel(size - unfilled_items > ptr_->reserved() ? size - unfilled_items
463 : ptr_->reserved());
464 for (size_t i = unfilled_items; i < size; i++) {
465 fill_panel(ptr[i]);
466 }
467 } else {
468 for (size_t i = 0; i < size; i++) {
469 fill_panel(ptr[i]);
470 }
471 }
