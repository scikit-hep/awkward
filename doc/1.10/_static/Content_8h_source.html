134  py::object
135  builder_snapshot(const ak::BuilderPtr builder) {
136  ::NumpyBuffersContainer container;
137  int64_t form_key_id = 0;
138  std::string form = builder.get()->to_buffers(container, form_key_id);
139  py::dict kwargs;

141  kwargs[py::str("length")] = py::int_(builder.get()->length());
142  kwargs[py::str("container")] = container.container();
143  kwargs[py::str("key_format")] = py::str("{form_key}-{attribute}");
