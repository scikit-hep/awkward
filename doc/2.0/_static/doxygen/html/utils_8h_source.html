216  type_to_form(int64_t form_key_id) {
217  if (std::string(typeid(T).name()).find("awkward") != std::string::npos) {
218  return std::string("awkward type");
219  }
220 
221  std::stringstream form_key;
222  form_key << "node" << (form_key_id++);
223 
224  if (std::is_arithmetic<T>::value) {
225  std::string parameters(type_to_name<T>() + "\", ");
226  if (std::is_same<T, char>::value) {
227  parameters = std::string(
228  "uint8\", \"parameters\": { \"__array__\": \"char\" }, ");
229  }
230  return "{\"class\": \"NumpyArray\", \"primitive\": \"" + parameters +
231  "\"form_key\": \"" + form_key.str() + "\"}";
232  } else if (is_specialization<T, std::complex>::value) {
233  return "{\"class\": \"NumpyArray\", \"primitive\": \"" +
234  type_to_name<T>() + "\", \"form_key\": \"" + form_key.str() +
235  "\"}";
236  }
237 
238  typedef typename T::value_type value_type;
239 
240  if (is_iterable<T>) {
241  std::string parameters("");
242  if (std::is_same<value_type, char>::value) {
243  parameters =
244  std::string(" \"parameters\": { \"__array__\": \"string\" }, ");
245  }
246  return "{\"class\": \"ListOffsetArray\", \"offsets\": \"i64\", "
247  "\"content\":" +
248  type_to_form<value_type>(form_key_id) + ", " + parameters +
249  "\"form_key\": \"" + form_key.str() + "\"}";
